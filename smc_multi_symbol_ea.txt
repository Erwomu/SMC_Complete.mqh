// Parse comma-separated symbol list
bool ParseSymbolList(string symbolList, string &symbols[], int &count)
{
    // Remove spaces
    StringReplace(symbolList, " ", "");
    
    // Split by comma
    string sep = ",";
    ushort u_sep = StringGetCharacter(sep, 0);
    int splits = StringSplit(symbolList, u_sep, symbols);
    
    if(splits <= 0)
    {
        // If no commas, assume single symbol
        ArrayResize(symbols, 1);
        symbols[0] = symbolList;
        count = 1;
    }
    else
    {
        count = splits;
    }
    
    // Validate count
    if(count > Max_Symbols)
    {
        Print("Symbol count exceeds maximum: ", Max_Symbols);
        count = Max_Symbols;
        ArrayResize(symbols, count);
    }
    
    return count > 0;
}

// Monitor open positions
void MonitorPositions()
{
    for(int i = PositionsTotal() - 1; i >= 0; i--)
    {
        CPositionInfo position;
        if(!position.SelectByIndex(i)) continue;
        
        if(position.Magic() != Magic_Number) continue;
        
        // Calculate position metrics
        double profit = position.Profit();
        double points = 0;
        
        if(position.PositionType() == POSITION_TYPE_BUY)
        {
            points = (position.PriceCurrent() - position.PriceOpen()) / 
                    SymbolInfoDouble(position.Symbol(), SYMBOL_POINT);
        }
        else
        {
            points = (position.PriceOpen() - position.PriceCurrent()) / 
                    SymbolInfoDouble(position.Symbol(), SYMBOL_POINT);
        }
        
        // Optional: Implement trailing stop or breakeven logic here
        if(points > 50) // Example: 50 points in profit
        {
            // Move stop to breakeven
            double newSL = position.PriceOpen();
            if(position.PositionType() == POSITION_TYPE_SELL)
            {
                if(newSL < position.StopLoss() || position.StopLoss() == 0)
                {
                    trade.PositionModify(position.Ticket(), newSL, position.TakeProfit());
                }
            }
            else
            {
                if(newSL > position.StopLoss() || position.StopLoss() == 0)
                {
                    trade.PositionModify(position.Ticket(), newSL, position.TakeProfit());
                }
            }
        }
    }
}

// Update trade statistics
void UpdateTradeStatistics()
{
    static int lastHistoryTotal = 0;
    int currentHistoryTotal = HistoryDealsTotal();
    
    if(currentHistoryTotal != lastHistoryTotal)
    {
        // Recalculate statistics
        g_winningTrades = 0;
        g_losingTrades = 0;
        
        HistorySelect(0, TimeCurrent());
        
        for(int i = HistoryDealsTotal() - 1; i >= 0; i--)
        {
            ulong ticket = HistoryDealGetTicket(i);
            if(ticket <= 0) continue;
            
            if(HistoryDealGetInteger(ticket, DEAL_MAGIC) != Magic_Number) continue;
            if(HistoryDealGetInteger(ticket, DEAL_ENTRY) != DEAL_ENTRY_OUT) continue;
            
            double profit = HistoryDealGetDouble(ticket, DEAL_PROFIT);
            
            if(profit > 0) g_winningTrades++;
            else if(profit < 0) g_losingTrades++;
        }
        
        lastHistoryTotal = currentHistoryTotal;
    }
}

{

// Check if trading is allowed
bool IsTradeAllowed()
{
    // Check if auto trading is enabled
    if(!TerminalInfoInteger(TERMINAL_TRADE_ALLOWED))
    {
        Print("Auto trading is disabled in terminal");
        return false;
    }
    
    // Check if EA trading is allowed
    if(!MQLInfoInteger(MQL_TRADE_ALLOWED))
    {
        Print("EA trading is not allowed");
        return false;
    }
    
    // Check account trade mode
    ENUM_ACCOUNT_TRADE_MODE tradeMode = (ENUM_ACCOUNT_TRADE_MODE)AccountInfoInteger(ACCOUNT_TRADE_MODE);
    if(tradeMode == ACCOUNT_TRADE_MODE_DEMO)
    {
        // Demo account - trading allowed
        return true;
    }
    else if(tradeMode == ACCOUNT_TRADE_MODE_CONTEST)
    {
        // Contest account - trading allowed
        return true;
    }
    else if(tradeMode == ACCOUNT_TRADE_MODE_REAL)
    {
        // Real account - check additional conditions
        if(!AccountInfoInteger(ACCOUNT_TRADE_ALLOWED))
        {
            Print("Trading not allowed on this account");
            return false;
        }
    }
    
    return true;
}

// Calculate daily profit
double CalculateDailyProfit()
{
    double dailyProfit = 0;
    
    datetime dayStart = iTime(_Symbol, PERIOD_D1, 0);
    HistorySelect(dayStart, TimeCurrent());
    
    for(int i = HistoryDealsTotal() - 1; i >= 0; i--)
    {
        ulong ticket = HistoryDealGetTicket(i);
        if(ticket <= 0) continue;
        
        if(HistoryDealGetInteger(ticket, DEAL_MAGIC) != Magic_Number) continue;
        if(HistoryDealGetInteger(ticket, DEAL_ENTRY) != DEAL_ENTRY_OUT) continue;
        
        dailyProfit += HistoryDealGetDouble(ticket, DEAL_PROFIT);
        dailyProfit += HistoryDealGetDouble(ticket, DEAL_SWAP);
        dailyProfit += HistoryDealGetDouble(ticket, DEAL_COMMISSION);
    }
    
    return dailyProfit;
}

// Risk management check
bool RiskManagementCheck()
{
    // Check drawdown limit
    double equity = AccountInfoDouble(ACCOUNT_EQUITY);
    double balance = AccountInfoDouble(ACCOUNT_BALANCE);
    double drawdown = (balance - equity) / balance * 100;
    
    if(drawdown > 20) // 20% drawdown limit
    {
        Print("WARNING: Drawdown exceeds 20% - ", DoubleToString(drawdown, 2), "%");
        return false;
    }
    
    // Check daily loss limit
    double dailyProfit = CalculateDailyProfit();
    double dailyLossLimit = balance * 0.05; // 5% daily loss limit
    
    if(dailyProfit < -dailyLossLimit)
    {
        Print("Daily loss limit reached: ", DoubleToString(dailyProfit, 2));
        return false;
    }
    
    // Check margin level
    double marginLevel = AccountInfoDouble(ACCOUNT_MARGIN_LEVEL);
    if(marginLevel > 0 && marginLevel < 200)
    {
        Print("WARNING: Low margin level - ", DoubleToString(marginLevel, 2), "%");
        return false;
    }
    
    return true;
}

// Calculate lot size based on risk percentage
double CalculateLotSize(string symbol, double riskPercent, double stopLossPoints)
{
    if(stopLossPoints <= 0) return 0.01; // Minimum lot
    
    double accountBalance = AccountInfoDouble(ACCOUNT_BALANCE);
    double riskAmount = accountBalance * (riskPercent / 100.0);
    
    double tickValue = SymbolInfoDouble(symbol, SYMBOL_TRADE_TICK_VALUE);
    double tickSize = SymbolInfoDouble(symbol, SYMBOL_TRADE_TICK_SIZE);
    double point = SymbolInfoDouble(symbol, SYMBOL_POINT);
    
    if(tickValue <= 0 || tickSize <= 0) return 0.01;
    
    double stopLossValue = stopLossPoints * point;
    double lotSize = riskAmount / (stopLossValue / tickSize * tickValue);
    
    // Normalize lot size
    double minLot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MIN);
    double maxLot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MAX);
    double lotStep = SymbolInfoDouble(symbol, SYMBOL_VOLUME_STEP);
    
    lotSize = MathMax(minLot, lotSize);
    lotSize = MathMin(maxLot, lotSize);
    lotSize = MathRound(lotSize / lotStep) * lotStep;
    
    return NormalizeDouble(lotSize, 2);
}

// Calculate Sharpe Ratio (simplified)
double CalculateSharpeRatio()
{
    if(g_totalTrades < 20) return 0; // Need minimum trades
    
    double returns[];
    ArrayResize(returns, g_totalTrades);
    
    HistorySelect(0, TimeCurrent());
    int index = 0;
    
    for(int i = 0; i < HistoryDealsTotal() && index < g_totalTrades; i++)
    {
        ulong ticket = HistoryDealGetTicket(i);
        if(ticket <= 0) continue;
        
        if(HistoryDealGetInteger(ticket, DEAL_MAGIC) != Magic_Number) continue;
        if(HistoryDealGetInteger(ticket, DEAL_ENTRY) != DEAL_ENTRY_OUT) continue;
        
        returns[index++] = HistoryDealGetDouble(ticket, DEAL_PROFIT);
    }
    
    if(index == 0) return 0;
    
    // Calculate average return
    double avgReturn = 0;
    for(int i = 0; i < index; i++)
        avgReturn += returns[i];
    avgReturn /= index;
    
    // Calculate standard deviation
    double variance = 0;
    for(int i = 0; i < index; i++)
        variance += MathPow(returns[i] - avgReturn, 2);
    variance /= index;
    
    double stdDev = MathSqrt(variance);
    
    if(stdDev == 0) return 0;
    
    return avgReturn / stdDev;
}

// Calculate profit factor
double CalculateProfitFactor()
{
    double grossProfit = 0;
    double grossLoss = 0;
    
    HistorySelect(0, TimeCurrent());
    
    for(int i = 0; i < HistoryDealsTotal(); i++)
    {
        ulong ticket = HistoryDealGetTicket(i);
        if(ticket <= 0) continue;
        
        if(HistoryDealGetInteger(ticket, DEAL_MAGIC) != Magic_Number) continue;
        if(HistoryDealGetInteger(ticket, DEAL_ENTRY) != DEAL_ENTRY_OUT) continue;
        
        double profit = HistoryDealGetDouble(ticket, DEAL_PROFIT);
        
        if(profit > 0)
            grossProfit += profit;
        else
            grossLoss += MathAbs(profit);
    }
    
    if(grossLoss == 0) return grossProfit > 0 ? 999.99 : 0;
    
    return grossProfit / grossLoss;
}
